# The Vulkan render system compiles its GLSL shaders through glslang,
# image codecs (png, ...) are plugins since Ogre 1.11
set(QSim_OGRE_PLUGINS ${QSim_OGRE_RENDER_SYSTEM} Plugin_GLSLangProgramManager Codec_STBI)

# plugins.cfg for running straight from the build directory
foreach(_config Release Debug)
  string(TOLOWER ${_config} _config_lower)
  set(_plugins_cfg "${CMAKE_BINARY_DIR}/bin_${_config_lower}/plugins.cfg")
  if(_config STREQUAL "Debug")
    set(_plugins_cfg "${CMAKE_BINARY_DIR}/bin_${_config_lower}/plugins_d.cfg")
  endif()
  set(_plugins_content "# Generated by CMake, plugins of the OGRE build q-sim was configured with\nPluginFolder=${OGRE_PLUGIN_DIR}\n")
  foreach(_plugin ${QSim_OGRE_PLUGINS})
    string(APPEND _plugins_content "Plugin=${_plugin}\n")
  endforeach()
  file(GENERATE OUTPUT ${_plugins_cfg} CONTENT "${_plugins_content}")
endforeach()
